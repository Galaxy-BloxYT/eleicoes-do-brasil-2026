local function safeDecode(raw)
    if type(raw) ~= "string" or raw == "" or raw == "null" then return nil end
    local ok, data = pcall(function() return HttpService:JSONDecode(raw) end)
    if not ok then warn("[TSE] Invalid JSON response:", tostring(data)); return nil end
    return data
end

-- VERIFICAÇÃO DE REGRAS JÁ ACEITAS (ARQUIVO LOCAL .JSON)
-- =========================================================
local ARQUIVO_REGRAS = "tse_regras_aceitas.json"
local function jaAceitouRegras()
    if not isfile or not readfile then return false end
    local sucesso, conteudo = pcall(function()
        return readfile(ARQUIVO_REGRAS)
    end)
    if sucesso and conteudo then
        local ok, dados = pcall(function() return HttpService:JSONDecode(conteudo) end)
        if ok and type(dados) == "table" and dados.aceito == true then return true end
    end
    return false
end

                    local votosGerais = type(dadosJson.v) == "table" and dadosJson.v or {}
                    local eleitorado = type(dadosJson.e) == "table" and dadosJson.e or {}
                    local function readNumber(obj, key)
                        if type(obj) ~= "table" then return nil end
                        local value = obj[key]
                        if value == nil then return nil end
                        local normalized = tostring(value):gsub("%s+", ""):gsub(",", ".")
                        return tonumber(normalized)
                    end
                    local function displayNumber(n)
                        return n ~= nil and formatarNumero(math.floor(n)) or "—"
                    end
                    local function displayPercent(value)
                        if value == nil or tostring(value) == "" then return "—" end
                        return tostring(value):gsub("%.", ",") .. "%"
                    end
                    local valid = readNumber(votosGerais, "vv")
                    local nullVotes = readNumber(votosGerais, "tvn") or readNumber(votosGerais, "vn")
                    local blank = readNumber(votosGerais, "vb")
                    local total = readNumber(votosGerais, "tv")
                    local computed = readNumber(votosGerais, "vvc")
                    local pctValid = votosGerais.pvvc
                    local pctNull = votosGerais.ptvn
